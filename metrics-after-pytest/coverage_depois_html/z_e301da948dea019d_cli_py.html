14#: The file read at startup, in the directory the command is run from.

16#: ⛔ CWD ONLY, not a search upwards. `find_dotenv` walks parent directories, 

17#: which means running the command from a subfolder can silently pick up 

18#: somebody else's file - a different key, a different browser - and nothing on 

19#: screen says which one was used. One predictable location is worth more than 

20#: the convenience. 

21ENV_FILE = ".env"

24def load_env_file(directory=None) -> dict: 

25 """Read `.env` from `directory` (default: the current one) into the process. 

26 

27 ⛔ IT NEVER OVERRIDES A VARIABLE THAT IS ALREADY SET, and that ordering is 

28 the whole design. A variable exported in the shell is something the user 

29 just did; a line in a file is something they did once, weeks ago. The recent 

30 decision wins, so the precedence a reader can rely on is: 

31 

32 --flag > the environment > .env > the default 

33 

34 Returns what it actually applied, so the caller can say so rather than 

35 leaving the user to guess whether the file was found at all. 

36 """ 

37 from pathlib import Path 

38 

39 path = Path(directory or Path.cwd()) / ENV_FILE 

40 if not path.is_file(): 

41 return {} 

42 

43 # dotenv rather than a hand-rolled parser: quoting, `export ` prefixes, 

44 # comments and multi-line values are all things a hand-rolled one gets 

45 # wrong on somebody else's machine, months later. 

46 from dotenv import dotenv_values 

47 

48 applied = {} 

49 for name, value in dotenv_values(path).items(): 

50 if value is None or name in os.environ: 

51 continue 

52 os.environ[name] = value 

53 applied[name] = value 

54 return applied

73class _EngineLine: 

74 """One terminal line that follows the download, redrawn in place. 

75 

76 On a terminal every whole percent is drawn; anywhere else (a log, a pipe) 

77 every tenth, because a carriage return in a file is not a redraw and a 

78 thousand of them is a wall of text. 

79 """ 

80 

81 def __init__(self, echo, tty: bool) -> None: 

82 self._echo = echo 

83 self._step = 1 if tty else 10 

84 self._drawn = None 

85 

86 def _draw(self, text: str) -> None: 

87 self._echo("\rbrowser %-40s" % text, nl=False) 

88 

89 def progress(self, done: int, total: int) -> None: 

90 pct = int(done * 100 / total) if total else 0 

91 bucket = pct // self._step 

92 if bucket == self._drawn: 

93 return 

94 self._drawn = bucket 

95 self._draw("downloading %3d%% %d MB" % (pct, done // (1 << 20))) 

96 

97 def status(self, phase: str) -> None: 

98 if phase != "downloading": 

99 self._draw(phase) 

100 

101 def done(self) -> None: 

102 self._draw("ready") 

103 self._echo("")

106def engine_on_disk(binary, echo=click.echo, fetch=None, tty=None) -> None: 

107 """Have the browser on disk before anything is served. 

108 

109 The servers this interface spawns would download it themselves, one per 

110 conversation, and from here that moment used to be invisible: this command 

111 printed "server connected" while nothing had been downloaded, and the 

112 first message then sat for a minute with nothing moving. So the download 

113 happens in front of the person, on one line that follows it, before the 

114 port opens; the spawned servers then find the cache. A given --binary 

115 skips it here the way it skips the download everywhere else; the server 

116 still checks that binary against the seal when it launches. 

117 """ 

118 from .engine import FETCH_BY_HAND, Engine 

119 

120 if binary: 

121 echo("browser %s" % binary) 

122 return 

123 line = _EngineLine(echo, tty=sys.stdout.isatty() if tty is None else tty) 

124 engine = Engine(fetch=fetch, listener=line) 

125 if engine.run() is None: 

126 echo("") 

127 raise click.ClickException( 

128 "the engine download failed: %s. Run it again, or by hand: %s" 

129 % (engine.error, FETCH_BY_HAND)) 

130 line.done()

133@click.group(invoke_without_command=True) 

134@click.pass_context 

135def main(ctx) -> None: 

136 """Drive a stealth browser with an LLM. 

137 

138 Without a subcommand this is the MCP server over stdio: `uvx invisible-playwright-mcp` is 

139 what an assistant registers, `python -m invisible_playwright_mcp` is what the interface 

140 spawns. `invisible-playwright-mcp ui` is the interface, which brings a model. 

141 

142 The model comes from OpenRouter and nowhere else: pass --openrouter-key or 

143 set OPENROUTER_API_KEY. Without one the interface refuses to start - an 

144 agent is a model with a browser, and there is no half of it to serve. To 

145 drive the browser by hand without a model, use the invisible_playwright 

146 library directly: same engine, Playwright's whole API. 

147 

148 Said here rather than only in the subcommand because this is the first page 

149 anybody reads, and a key requirement discovered from an error message is a 

150 key requirement discovered too late. 

151 

152 A `.env` in the directory you run from is read first, so the key and the 

153 browser path can live in a file instead of a shell profile. It never 

154 overrides something already in the environment. 

155 

156 Serving as the MCP server, this process then drops the model key from its 

157 own environment whatever it came from, because a browser server has no use 

158 for one and the engine it launches inherits what this holds. 

159 """ 

160 applied = load_env_file() 

161 serving = ctx.invoked_subcommand is None 

162 

163 # ⛔ AFTER `.env`, BEFORE ANYTHING READS A SESSION. The file can be where 

164 # the home override is set, and the move must not happen when it is. One 

165 # startup edge, once per process: `storage.home()` stays a pure answer to 

166 # "where is the directory" and this is the one place that decides about 

167 # somebody's data. It says so, because a directory moving under a person 

168 # with nothing printed is the same silence as losing it. 

169 carried = storage.carry_over_the_old_directory() 

170 if carried is not None: 

171 click.echo("sessions moved to %s, from the directory the previous name " 

172 "used: nothing was merged and nothing was deleted" % carried, 

173 err=serving) 

174 if serving: 

175 # ⛔ A BROWSER SERVER HAS NO USE FOR A MODEL KEY, AND WHATEVER THIS 

176 # PROCESS HOLDS THE ENGINE INHERITS. This is where 0.68.2 handed it 

177 # back: the interface strips the key from the environment it gives the 

178 # child, carefully and with twenty-two tests behind it, and then the 

179 # child runs this very function, finds the key in `.env`, and puts it 

180 # straight back. The line below printed it twice, once per process, and 

181 # that was the whole visible evidence. 

182 # 

183 # Read from the environment rather than filtered out of the file, so it 

184 # covers the same key arriving any other way - exported in the shell by 

185 # somebody running `uvx invisible-playwright-mcp` by hand, or under an alias. The reason 

186 # is on `runner.forget_key`. 

187 # 

188 # The names are dropped from what gets REPORTED too: saying a file 

189 # applied something this process then threw away is a line that is not 

190 # true by the time anybody reads it. 

191 for name in forget_key(os.environ): 

192 applied.pop(name, None) 

193 if applied: 

194 # Names, never values: this line exists so a reader knows the file was 

195 # found, and printing what was in it would put the key on the terminal. 

196 # On stderr when serving, because stdout is then the protocol channel. 

197 click.echo("env %s: %s" % (ENV_FILE, ", ".join(sorted(applied))), 

198 err=serving) 

199 if serving: 

200 _serve()

226def ui(openrouter_key, model, host, port, proxy, seed, headed, binary, profile_dir): 

227 """Serve the two-pane interface: conversation left, live browser right. 

228 

229 Requires an OpenRouter key: an agent is a model with a browser, and 

230 without the model there is nothing honest to serve. Driving the browser 

231 by hand, no model and nothing spent, is the invisible_playwright 

232 library's job - same engine, Playwright's whole API. 

233 """ 

234 from .agent import OpenRouterBrain 

235 from .chat import DEFAULT_CHAT_ID 

236 from .llm import make_client 

237 from .routes import build_app 

238 from .sessions import Sessions

240 # ⛔ THE RULE LIVES IN `llm.resolve_key`, AND UNTIL NOW THIS RE-IMPLEMENTED 

241 # IT. Both said the same thing - the flag beats the variable, an empty one 

242 # counts as absent, nothing at all is a refusal - which is exactly the 

243 # shape that goes wrong quietly: `resolve_key` carried a dozen assertions 

244 # pinning those edges and had no caller in the product, so the tested copy 

245 # and the running copy were different code. Either could have drifted 

246 # without a red test. 

247 # 

248 # What stays here is the PRESENTATION, which is genuinely the CLI's: a 

249 # ClickException prints one line and exits 1 where a RuntimeError dumps a 

250 # traceback, and the sentence names the library for somebody who wanted a 

251 # browser rather than an agent. 

252 try: 

253 key = resolve_key(openrouter_key, os.environ) 

254 except RuntimeError: 

255 raise click.ClickException( 

256 "no OpenRouter key. Pass --openrouter-key or set " 

257 "OPENROUTER_API_KEY. To drive the browser without a model, use " 

258 "the invisible_playwright library directly: same engine, " 

259 "Playwright's whole API.") 

260 mdl = resolve_model(model, os.environ) 

261 # ONE client, a brain PER CONVERSATION. The client is a connection and the 

262 # brain is a transcript: sharing the first is what it is for, and sharing 

263 # the second would give every session in the column the same memory, so 

264 # asking one thing in a session would answer with another session's work. 

265 client = make_client(key) 

266 click.echo("model %s via %s" % (mdl, BASE_URL)) 

267 # After the key, so a refusal costs nobody a quarter-gigabyte download; 

268 # before the link, so the spawned servers find the engine on disk. 

269 engine_on_disk(binary) 

270 

271 opts = {"proxy": proxy, "seed": seed, "headed": headed, 

272 "binary": binary, "profile_dir": profile_dir} 

273 

274 async def serve() -> None: 

275 import uvicorn 

276 

277 sessions = Sessions(opts, key, lambda: OpenRouterBrain(client, mdl), 

278 model_label=mdl) 

279 # ⛔ THE DEFAULT CONVERSATION IS STARTED EAGERLY, EVERY OTHER ONE 

280 # LAZILY. Every conversation spawns its own server now, on first use - 

281 # `Sessions.get` - and the interface used to open ONE connection at 

282 # boot just to prove the server starts and to report its tool count. 

283 # Asking for `default` here reproduces exactly that boot experience: 

284 # it is the conversation almost every install actually opens first, 

285 # and its connection is real rather than a throwaway diagnostic one. 

286 default = await sessions.get(DEFAULT_CHAT_ID) 

287 click.echo("server connected, %d tools" % len(default.link.tools)) 

288 click.echo("open http://%s:%d" % (host, port)) 

289 app = build_app(sessions) 

290 server = uvicorn.Server(uvicorn.Config(app, host=host, port=port, 

291 log_level="warning")) 

292 try: 

293 await server.serve() 

294 finally: 

295 await sessions.close_all() 

296 

297 try: 

298 asyncio.run(serve()) 

299 except KeyboardInterrupt: 

300 pass
