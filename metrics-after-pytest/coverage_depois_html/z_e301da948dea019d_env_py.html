1"""What this package reads from the environment, and what it still answers to. 

2 

3⛔ THE NAMES LIVE HERE AND NOWHERE ELSE, BECAUSE FROM 2026-09-23 EACH OF THEM IS 

4TWO STRINGS: the name it has, and the name it had. Before that they were one 

5string each and a literal at the single place that read it was the right shape. 

6Two strings each is not: three modules read one of these and a fourth writes 

7one, so the pair would be six literals, and a rename that reaches five of them 

8is the silent kind. One house, and every caller asks it. 

9 

10The old prefix was the product's previous brand. It was kept through the package 

11rename on the reasoning that somebody who had set it would otherwise find it 

12IGNORED, which is the worst way to break a configuration; the owner then asked 

13for the name to be abandoned outright. Both things are true at once, and this 

14module is how: the new name is the only one the product emits, documents or 

15teaches, and the old one still ANSWERS, out loud. 

16 

17Precedence, when both are set: the new name wins. It is the more recent 

18decision, the same reasoning that makes the shell beat `.env` in `cli.py`. 

19 

20`read` says so through the logger rather than a print, on purpose. stdout is the 

21MCP protocol channel and a print there corrupts the stream; the logger's default 

22destination is stderr, which a client discards and a person at a terminal sees. 

23Once per name per process, because a warning repeated on every lookup is a 

24warning people filter out. 

25""" 

26from __future__ import annotations 

27 

28import logging 

29import os 

30from typing import Mapping, Optional 

31 

32log = logging.getLogger("invisible_playwright_mcp") 

33 

34#: Where sessions are kept, when the default place is not wanted. 

35HOME = "INVISIBLE_MCP_HOME" 

36#: The OpenRouter model id the interface and the agent use. 

37MODEL = "INVISIBLE_MCP_MODEL" 

38#: Which saved session this process serves. Set by `runner.child_env` for a 

39#: server it spawns, and read once at import by `mcp.server`. 

40SESSION_ID = "INVISIBLE_MCP_SESSION_ID" 

41 

42#: new name -> the name it replaced on 2026-09-23. A name is in this map for 

43#: exactly as long as somebody might still have it set; removing an entry is a 

44#: deliberate act, and the entry is what makes that act visible. 

45RETIRED = { 

46 HOME: "AIHAWK_HOME", 

47 MODEL: "AIHAWK_MODEL", 

48 SESSION_ID: "AIHAWK_SESSION_ID", 

49} 

50 

51#: Names already warned about, so the notice is one per process and not one per 

52#: lookup. Module state on purpose: the point is that it outlives the call. 

53#: 

54#: ⛔ IT HAD A `forget_what_was_said()` BESIDE IT FOR ONE RUN, so a test could 

55#: reset it without touching a private name. The surface gate refused it, and 

56#: the gate was right: a function in the product that nothing in the product 

57#: calls is surface somebody will later mistake for a feature. The test that 

58#: asserts on the deduplication clears this directly, which is fair - the 

59#: deduplication is its whole subject. 

60_SAID: set = set() 

61 

62 

63def read(name: str, env: Optional[Mapping[str, str]] = None) -> Optional[str]: 

64 """The value set under `name`, or under the name it replaced, or None. 

65 

66 `env` is a mapping so a test can hand one in; it defaults to the process's. 

67 """ 

68 where = os.environ if env is None else env 

69 value = where.get(name) 

70 if value: 

71 return value 

72 

73 was = RETIRED.get(name) 

74 if not was: 

75 return None 

76 old = where.get(was) 

77 if not old: 

78 return None 

79 

80 if was not in _SAID: 

81 _SAID.add(was) 

82 log.warning( 

83 "%s is the retired name for %s and is what answered. It still works; " 

84 "the new name is what this package reads first, documents and will " 

85 "keep. Rename it in your shell or your .env when convenient.", 

86 was, name) 

87 return old
