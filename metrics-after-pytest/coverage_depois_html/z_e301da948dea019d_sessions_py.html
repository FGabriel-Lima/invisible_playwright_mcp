31class Sessions: 

32 """Every conversation this interface holds, by id, saved as it goes. 

33 

34 ⛔ A CONVERSATION AND ITS BROWSERS ARE ONE SESSION, and this class is where 

35 that is true rather than nearly true - but what makes it true changed on 

36 2026-09-11. It used to be one shared MCP connection with the id imposed on 

37 every call, because MCP itself understood "session" as a tool argument. It 

38 does not any more: no tool takes one, none can be listed, none can be 

39 reached from another. So the id can no longer be a value passed over an 

40 open connection - it has to be which CONNECTION exists at all, and that 

41 means one conversation, one spawned server PROCESS, its `INVISIBLE_MCP_SESSION_ID` 

42 set at the moment it is started. Two ids would have been easier only in the 

43 sense that it used to be true; it is not the shape MCP has any more. 

44 

45 Conversations are built ON DEMAND and read from disk the first time they are 

46 asked for, which now means spawning their process the first time too - the 

47 same lazy cost `browser_open` already pays for a browser, paid once more, one 

48 layer up, for the connection that reaches it. They are not all loaded at 

49 startup: a transcript is thousands of lines and somebody with twenty 

50 sessions wants a column of names, not twenty processes to draw it. 

51 """ 

52 

53 def __init__(self, opts: Mapping[str, Any], key: Optional[str], make_brain, 

54 model_label: str = "no model", *, open_link=None) -> None: 

55 self._opts = dict(opts) 

56 self._key = key 

57 self._make_brain = make_brain 

58 self.model_label = model_label 

59 self._live: Dict[str, ChatService] = {} 

60 # ⛔ HOW A CONNECTION IS MADE, AS A PARAMETER RATHER THAN AS AN 

61 # ATTRIBUTE TO OVERWRITE. It was `self._open_link = self._spawn_link`, 

62 # reassigned from outside by two test modules - a seam that worked and 

63 # that nothing declared: a reader of the signature could not see it, 

64 # and a reader of the assignment could not tell a convention from an 

65 # accident. Passed in, it is part of the interface and the default is 

66 # the product's one and only way. 

67 # 

68 # What it is for is unchanged: a test that wants many conversations 

69 # without many real processes supplies its own, so what is under test 

70 # is `Sessions` deciding WHICH conversation gets WHICH connection - 

71 # never `Link` itself, which has its own tests, and never a real 

72 # subprocess, which `tests/mcp_server/test_stdio_e2e.py` proves. 

73 self._open_link = open_link or self._spawn_link 

74 #: ⛔ ONE CONVERSATION IS MADE AT A TIME, AND THE COST OF THAT IS THE 

75 #: POINT. Making one is the slow thing here - a process starts and 

76 #: shakes hands - and it was done OUTSIDE any guard, so every request 

77 #: that arrived while one was being made started another. See `get`. 

78 #: 

79 #: One lock and not one per id: the fast path, a conversation already 

80 #: open, never touches it, so what this serialises is the making of two 

81 #: conversations at the same instant. That is rare - a page opening, a 

82 #: session being created - and it is already expensive, while a 

83 #: dictionary of locks is a second registry that has to be kept in 

84 #: agreement with `_live` and emptied by somebody. 

85 self._making = asyncio.Lock()

87 async def _spawn_link(self, session_id: str) -> Link: 

88 return await Link(dict(self._opts, session_id=session_id), 

89 key=self._key).open()

91 # ⛔ `around` STOOD HERE: A CLASSMETHOD WITH ELEVEN CALLERS, NONE OF THEM 

92 # THE PRODUCT. It built a registry holding one conversation somebody else 

93 # had made, and it justified itself with "the tests do, and so would 

94 # anything embedding this" - a user that does not exist. This is an 

95 # application; the only importer of this module is `cli.py`. 

96 # 

97 # The argument it was written for was good and it did not move with it: 

98 # a test driving ONE conversation should still go through `build_app` and 

99 # the routes, so the single case is exercised by the same code the many 

100 # case uses. That is a rule about the suite, and it now lives in the suite, 

101 # in `tests/_sessions.py`. What is left here is the constructor the product 

102 # calls and nothing else.

104 async def close_all(self) -> None: 

105 """Close every conversation's own connection, for a clean shutdown. 

106 

107 ⛔ THERE IS NO LONGER ONE CONNECTION TO CLOSE - there are as many as 

108 conversations were ever asked for, each its own process. `Link.close` 

109 already swallows what closing can throw, so nothing here needs to. 

110 """ 

111 for service in self._live.values(): 

112 await service.link.close()

114 async def get(self, session_id: str | None = None) -> ChatService: 

115 """The conversation with this id, loaded from disk the first time. 

116 

117 ⛔ SPAWNS ITS OWN SERVER, THE FIRST TIME, because that is now the only 

118 way one conversation's browsers stay apart from another's: each gets 

119 its own process, with `INVISIBLE_MCP_SESSION_ID` set to this id, so the server 

120 inside never has more than the one thing to persist and never anything 

121 to confuse it with. 

122 """ 

123 at = session_id or DEFAULT_CHAT_ID 

124 found = self._live.get(at) 

125 if found is not None: 

126 return found 

127 async with self._making: 

128 # ⛔ ASKED AGAIN, INSIDE. Whoever held the lock may have been making 

129 # exactly this conversation, and the answer that mattered was taken 

130 # before waiting for them. Without this second look the guard only 

131 # moves the race, it does not close it. 

132 found = self._live.get(at) 

133 if found is not None: 

134 return found 

135 return await self._make(at)

137 async def _make(self, at: str) -> ChatService: 

138 """Build the conversation with this id and register it. The caller 

139 holds `_making`, and that is not a detail: everything slow happens 

140 here, and `_live` is written at the end of it. 

141 """ 

142 link = await self._open_link(at) 

143 service = ChatService(link, self._make_brain(), 

144 model_label=self.model_label, session_id=at) 

145 service.restore() 

146 self._live[at] = service 

147 return service

150 """An id no conversation has. Read under `_making`, because a 

151 conversation being MADE is in neither place this looks.""" 

152 at = "s%d" % int(time.time() * 1000) 

153 while at in self._live or chats.load_chat(at) is not None: 

154 at += "x" 

155 return at

157 async def new(self) -> ChatService: 

158 """A conversation nobody has used yet, with an id of its own. 

159 

160 The id is the clock, not a counter: a counter has to be stored somewhere 

161 to survive a restart, and the place it would be stored is the thing that 

162 breaks. It is never shown - the name is - so it only has to be unique. 

163 

164 ⛔ THE ID IS CHOSEN AND CLAIMED WITHOUT LETTING GO, and that is the whole 

165 of this method. It used to pick an id, find it free, and then `await 

166 get(at)`: two callers in the same MILLISECOND computed the same id, both 

167 found it free - a conversation being made is in neither `_live` nor on 

168 disk - and the second was handed the first one's conversation. Measured: 

169 two `new()` at once returned the same object, so somebody who asked for 

170 a new conversation got somebody else's, with its transcript. 

171 

172 Holding `_making` across both halves is what makes "free" mean 

173 something, because the previous caller registers before releasing. 

174 """ 

175 async with self._making: 

176 return await self._make(self._free_id())

178 def listing(self) -> List[dict]: 

179 """Every conversation, saved or only live, newest first. 

180 

181 A conversation opened a moment ago has nothing on disk yet, and leaving 

182 it out would make the column disagree with the page it is drawn beside. 

183 """ 

184 rows = {r["id"]: dict(r) for r in chats.known_chats()} 

185 for at, service in self._live.items(): 

186 row = rows.setdefault(at, {"id": at, "saved": "", "turns": 0}) 

187 row["name"] = service.name 

188 row["turns"] = sum(1 for e in service.history if e.get("kind") == "you") 

189 row["live"] = True 

190 out = list(rows.values()) 

191 out.sort(key=lambda r: (r.get("saved") or "", r["id"]), reverse=True) 

192 return out

194 def knows(self, session_id: str | None) -> bool: 

195 """Whether this conversation exists, WITHOUT bringing it into being. 

196 

197 ⛔ THE QUESTION EVERY REQUEST HAS TO ASK BEFORE `get`. Building what it 

198 does not find is right for a caller that MEANS to start a conversation - 

199 `new`, an embedder, a test - and wrong for a request that only means to 

200 look at one. Measured: a page left open on a session somebody deleted 

201 went on asking `/live/browsers` every three seconds, and one of those 

202 questions was enough to declare the session again. The delete worked, 

203 answered `forgotten:true`, closed the browsers, erased the file - and the 

204 row came back on its own, empty and unnamed, for as long as that tab 

205 stayed open. Every visible signal said the delete had failed. 

206 

207 ⛔ AND IT ASKS ABOUT BOTH HALVES OF A SESSION, because a session is a 

208 conversation AND its browsers and either half can be the only one on 

209 disk. An agent client that opens a browser in session `work` and never 

210 touches this interface writes the browsers and no transcript; opening 

211 `?s=work` here has to work, and asking only about transcripts would have 

212 refused it. That is the same defect this method exists to fix, made one 

213 step further along - which is why it is written into the one question 

214 rather than into the routes that ask it. 

215 

216 Each half is asked of whoever owns it: what is in memory of this 

217 registry, what is on disk of the store. On disk means the FILE is 

218 there, not that it parses: `get` reads it once, right after, and 

219 reading it here as well was the same file parsed twice per request. The default is always known 

220 because it is the conversation a page with no id gets, and on a fresh 

221 install nothing has written it down yet. 

222 """ 

223 at = session_id or DEFAULT_CHAT_ID 

224 return (at == DEFAULT_CHAT_ID or at in self._live 

225 or chats.chat_path(at).is_file() 

226 or store.path_of(at).is_file())

237 async def forget(self, session_id: str) -> bool: 

238 """Delete a conversation AND the browsers that belonged to it. 

239 

240 ⛔ BOTH HALVES, BECAUSE THEY ARE ONE SESSION, AND HOW THE SECOND HALF 

241 HAPPENS CHANGED WITH EVERYTHING ELSE ON 2026-09-11. There used to be a 

242 tool, `session_forget`, that closed the browsers of a session named on 

243 an open connection shared with every other conversation. There is no 

244 such tool now, and there could not be one that stayed inside the rule 

245 the owner set: MCP cannot be asked to operate on a piece of work that 

246 is not its own. So this reaches for the ONLY thing that was ever 

247 really being asked for - closing that conversation's OWN connection, 

248 which is that conversation's OWN server process, and closing that ends 

249 it exactly the way a standalone client disconnecting always has: the 

250 stdio EOF this sends is what `_lifespan` in `mcp/server.py` already 

251 closes every browser on, the same path every such client relies on, 

252 not a new one built for this. 

253 

254 The saved browser file is erased directly, by calling into `store` in 

255 this same package - not over MCP, because MCP has no tool that reaches 

256 a piece of work other than its own, and this IS the process whose own 

257 file it is (or was, if it was never brought up this run). That call is 

258 a plain function, not a wire protocol: it costs nothing to make and 

259 asks nothing of anybody. 

260 

261 Refused while that conversation is mid-run: the same reason `reset` is. 

262 

263 ⛔ AND `False` MEANS THAT AND NOTHING ELSE. It used to mean "there was 

264 nothing to erase" as well, and the two are opposite news: the page reads 

265 it and says "that session is still working, so it was not deleted", which 

266 for a session somebody else had already deleted is the wrong sentence in 

267 both halves. The caller asked for it to be gone; if it is gone, the 

268 answer is yes. 

269 """ 

270 # ⛔ UNDER THE SAME LOCK AS THE MAKING, AND OUT OF THE REGISTRY BEFORE 

271 # THE CLOSING. Two windows, both the shape of §EM: this closed the 

272 # connection and only THEN dropped the service, so anything asking for 

273 # that conversation in between was handed a service whose link was 

274 # already shut; and it did not wait for a creation in flight, so a 

275 # conversation being made would register itself into `_live` after its 

276 # files had been erased - alive, unreachable by name, and holding a 

277 # process nobody would close. 

278 async with self._making: 

279 service = self._live.get(session_id) 

280 if service is not None and service.busy: 

281 return False 

282 self._live.pop(session_id, None) 

283 if service is not None: 

284 # Closes silently: `Link.close` already swallows what closing 

285 # can throw, because a connection being torn down on purpose is 

286 # not a failure to report back. 

287 await service.link.close() 

288 store.erase(session_id) 

289 chats.erase_chat(session_id) 

290 return True
