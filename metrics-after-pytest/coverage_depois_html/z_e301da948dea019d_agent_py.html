61def system_message(instructions: str = "") -> dict: 

62 """The one system message, from this build's prompt plus what the server 

63 says about itself. 

64 

65 ⛔ ONE FUNCTION, THREE WRITERS. The message was assembled in the 

66 constructor, in the brain's restore and nowhere else, and the server's 

67 instructions - the only text that defines what `support` is for and that 

68 whoever opens it closes it - were in neither. They travel with the link 

69 and are refreshed at the start of every run, so a restored transcript and 

70 a new one carry the same current instructions. 

71 

72 ⛔ AND THE TWO HALVES SAY DIFFERENT THINGS NOW. This prompt used to open 

73 with "open the browser with browser_open before anything else; if a tool 

74 answers that the browser is not open or is gone, call browser_open and 

75 carry on" - which is the server's first paragraph, in other words, glued 

76 to it a few lines later. The model read the same rule twice in one 

77 message, and two copies of a rule are two things to keep in agreement. 

78 

79 The rule belongs to the server: it is a fact about the tools, and a 

80 standalone client that never sees this prompt still has to be told it. 

81 What is left here is what this loop owns - how the agent behaves, and how 

82 it WRITES. 

83 

84 ⛔ AND MOVING IT MEANS THE SERVER MUST ACTUALLY STILL SAY IT, which is a 

85 thing to hold rather than to trust. A first draft of this change also made 

86 an empty `instructions` append a warning for the model to read; that was 

87 dropped, because the link launches this very server, so arriving here with 

88 nothing means the handshake failed and a sentence in the prompt helps 

89 nobody in that state. What guards the move instead is a gate over the 

90 server's own text: the rule has one home, and the gate says the home is 

91 not empty. 

92 """ 

93 text = SYSTEM_PROMPT 

94 if instructions: 

95 text += chr(10) + chr(10) + instructions.strip() 

96 return {"role": "system", "content": text}

131def mcp_tools_to_openai(tools) -> List[dict]: 

132 """MCP tool descriptions as OpenAI function definitions. 

133 

134 Two details are load-bearing and both come from the API rejecting the 

135 alternative: `parameters` must be an object and never None, and the 

136 description is truncated because a long one is rejected rather than trimmed. 

137 """ 

138 out = [] 

139 for t in tools: 

140 out.append({ 

141 "type": "function", 

142 "function": { 

143 "name": t.name, 

144 "description": (getattr(t, "description", "") or "")[:1024], 

145 "parameters": getattr(t, "inputSchema", None) or {"type": "object", "properties": {}}, 

146 }, 

147 }) 

148 return out

171def shorten(text: str, limit: int, where: str) -> str: 

172 """`text` cut to `limit`, saying so whenever it cuts. 

173 

174 ⛔ IT CUT IN SILENCE, AND THE PERSON WATCHING GOT THE SMALLER COPY. A read 

175 of a real page is tens of kilobytes; the page showed the first 1200 

176 characters, stopping mid-word with no ellipsis, no count and no hint that 

177 anything had been removed, while the line below handed the model nearly 

178 seven times as much. Somebody expanding a step to audit what the agent saw 

179 read a partial page as the whole one. The product's whole claim is that you 

180 can watch what it does, and the watcher was the one being given less. 

181 

182 Both callers say it now, because the model reading a truncated page without 

183 being told is the same defect one level up: it can ask for the rest, but 

184 only if it knows there is a rest. 

185 

186 The marker starts on a new line so the page files the result into an 

187 expandable block rather than onto the step row, and it is ASCII: a 

188 non-ASCII character on a line this server may print is a known way to kill 

189 the process on Windows. 

190 """ 

191 if len(text) <= limit: 

192 return text 

193 return "%s\n[... %d more characters, not %s]" % ( 

194 text[:limit], len(text) - limit, where)

210class Conversation: 

211 """One transcript, and the loop that grows it. 

212 

213 Kept as an object because the interface needs the transcript to survive an 

214 instruction: "and now sort them by price" only means something if the model 

215 still knows what "them" was. `do` throws the object away after one call and 

216 gets the old one-shot behaviour for free. 

217 """ 

218 

219 #: Ceiling on ONE reply, and it is set rather than left to the provider. 

220 #: Without it the provider assumes the model's maximum - 65536 on a current 

221 #: OpenAI model - and a credit-limited key is refused with a 402 before any 

222 #: work happens, quoting a token budget rather than naming the task. It is 

223 #: also the wrong shape for this loop: a turn is a sentence of reasoning and 

224 #: a tool call, not an essay, and the only turn that wants room is the last 

225 #: one. Generous for that, sixteen times smaller than the default. 

226 MAX_TOKENS = 8192 

227 

228 def __init__(self, client, model: str, *, 

229 max_tokens: int = MAX_TOKENS) -> None: 

230 self.client = client 

231 self.model = model 

232 self.max_tokens = max_tokens 

233 self.messages: List[dict] = [system_message()] 

234 self.tool_defs: Optional[List[dict]] = None 

235 self.usage = {"prompt": 0, "completion": 0, "calls": 0, "last_prompt": 0}

238 def known(self) -> List[str]: 

239 """The tool names this server has, read off the definitions. 

240 

241 ⛔ DERIVED, WHERE IT WAS A SECOND COPY ASSIGNED INSIDE A BRANCH. It was 

242 written in `run`, under `if self.tool_defs is None`, which is the one 

243 line that builds the definitions - so the list existed only when that 

244 branch had run, and it was the only attribute of this class not 

245 declared in `__init__`. A conversation whose definitions arrived any 

246 other way had no `known` at all, and the failure landed on the line 

247 below that tells a model it asked for a tool nobody has: an 

248 `AttributeError` in the handler for somebody else's mistake. 

249 

250 Initialising it to an empty list in `__init__` would have been worse 

251 than the crash, because an empty list is a legal answer: every tool the 

252 model asked for would be refused as unknown, quietly and wrongly. The 

253 fact is a property of `tool_defs`, so it is read from `tool_defs`. 

254 """ 

255 return [d["function"]["name"] for d in self.tool_defs or []]

257 def _note_usage(self, resp) -> None: 

258 u = getattr(resp, "usage", None) 

259 if u is None: 

260 return 

261 last = getattr(u, "prompt_tokens", 0) or 0 

262 self.usage["prompt"] += last 

263 self.usage["completion"] += getattr(u, "completion_tokens", 0) or 0 

264 self.usage["calls"] += 1 

265 # The LAST turn's prompt, kept beside the running totals and not folded 

266 # into them. Each turn is sent the whole transcript, so the newest prompt 

267 # size IS the current occupancy of the context window; adding them up 

268 # counts every earlier turn again and races past any limit within a few 

269 # messages, which would make a meter built on it worse than none. 

270 self.usage["last_prompt"] = last

272 async def run(self, task: str, call_tool, tools, *, say: Say = _silent, 

273 describe=None, instructions: str = "") -> str: 

274 """Run one instruction to an answer. 

275 

276 `call_tool(name, args)` performs a tool call and returns the MCP result; 

277 `tools` is the server's tool list. Both are passed rather than a session, 

278 so this has no opinion about how the browser is reached - which is what 

279 lets the interface serialise its calls against the frame pump. 

280 """ 

281 if self.tool_defs is None: 

282 self.tool_defs = mcp_tools_to_openai(tools) 

283 # Refreshed every run rather than set once: the instructions belong to 

284 # the server the link is talking to now, and a transcript restored from 

285 # disk arrived with a system message this process did not write. 

286 self.messages[0] = system_message(instructions) 

287 self.messages.append({"role": "user", "content": task}) 

288 

289 # No turn ceiling. There was one, and what it did in practice was end 

290 # long tasks that were going fine with "task did not finish within 

291 # max_turns=25" - a task the person had watched work for twenty-five 

292 # steps, thrown away one step before it might have answered, with the 

293 # transcript at its largest and every one of those steps already paid 

294 # for. A number cannot tell a loop that is stuck from a task that is 

295 # simply long, and guessing wrong costs the whole run. 

296 # 

297 # What stops a run instead is the person watching it: the interface 

298 # keeps the task handle and the send button becomes a stop button while 

299 # work is in flight, so a cancel lands at the next tool call. That is a 

300 # judgement about THIS run rather than a constant chosen in advance, 

301 # and unlike the ceiling it can also stop a run on turn three. 

302 while True: 

303 # IN A THREAD, and that is what makes the stop button work. The 

304 # client is synchronous, so called here it would occupy the event 

305 # loop for the whole request: measured at zero scheduler slices 

306 # during a 300 ms call, which means `/chat/stop` cannot be served, 

307 # no event reaches the page and the live pane does not repaint - 

308 # for the entire time the model is thinking, which is exactly when 

309 # somebody reaches for stop. 

310 # 

311 # It is also a cancellation point, so a stop lands here rather than 

312 # waiting for the turn to reach its next tool call. What it does 

313 # NOT do is unsend the request: the thread runs to completion and 

314 # its answer is dropped, so a run stopped mid-turn still pays for 

315 # the reply it never used. 

316 resp = await asyncio.to_thread( 

317 self.client.chat.completions.create, 

318 model=self.model, messages=self.messages, 

319 tools=self.tool_defs, tool_choice="auto", temperature=0, 

320 max_tokens=self.max_tokens, 

321 ) 

322 # Counted, saved with the transcript, and not announced: 

323 # the page drew it in a meter that is gone. The accounting 

324 # stays, because it is what the conversation cost; the 

325 # event went, because an event nobody draws is drawn as 

326 # raw JSON at the end of the transcript. 

327 self._note_usage(resp) 

328 msg = resp.choices[0].message 

329 self.messages.append(msg.model_dump()) 

330 

331 if getattr(msg, "content", None): 

332 await say("said", msg.content) 

333 if not msg.tool_calls: 

334 return msg.content or "" 

335 

336 for call in msg.tool_calls: 

337 name = call.function.name 

338 try: 

339 args = json.loads(call.function.arguments or "{}") 

340 except json.JSONDecodeError as exc: 

341 # Recoverable: telling the model its arguments were unreadable 

342 # lets it try again. Raising would end the whole task over one 

343 # malformed message. 

344 await say("err", f"{name}: unreadable arguments ({exc})") 

345 self.messages.append({"role": "tool", "tool_call_id": call.id, 

346 "content": f"arguments were not valid JSON: {exc}"}) 

347 continue 

348 

349 await say("tool", f"{name} {describe(name, args)}".strip() 

350 if describe else name) 

351 # ⛔ A TOOL THE MODEL REMEMBERS AND THIS SERVER DOES NOT HAVE. 

352 # Earlier public versions of this server had session_list, 

353 # session_start and session_status, and a model trained on 

354 # that repository reaches for them: measured 2026-09-13 with 

355 # the real model, `session_list` as the FIRST call in six runs 

356 # out of twelve, on a page that never mentions the word. The 

357 # server answered `Unknown tool: session_list` and nothing 

358 # else, so the model guessed again. The loop knows the real 

359 # list, so it answers here - no round trip - and says what 

360 # the model was looking for: nothing has to be listed or 

361 # started, the two browsers are already there. 

362 if name not in self.known: 

363 text = unknown_tool(name, self.known) 

364 await say("err", text) 

365 self.messages.append({"role": "tool", "tool_call_id": call.id, 

366 "content": text}) 

367 continue 

368 try: 

369 text, failed = answer_of(await call_tool(name, args)) 

370 except Exception as exc: 

371 text = f"{type(exc).__name__}: {exc}" 

372 await say("err", text) 

373 else: 

374 await say("err" if failed else "result", 

375 shorten(text, SHOWN, "shown")) 

376 self.messages.append({"role": "tool", "tool_call_id": call.id, 

377 "content": shorten(text, SENT, "sent")})

414
