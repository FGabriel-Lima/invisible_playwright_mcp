132 async def describe_pages(self) -> list[dict]: 

133 """Each page as url, title and whether it is the one a command drives. 

134 

135 Title costs a round trip per page and url does not, so a page that will 

136 not answer contributes what it can rather than failing the whole list - 

137 a page mid-navigation must not make the others unreadable. A closed 

138 target is the exception: see below. 

139 """ 

140 live = self.pages() 

141 out: list[dict] = [] 

142 for page in live: 

143 row = {"active": page is live[-1], "url": "", "title": ""} 

144 # ⛔ `page.url` IS CACHED AND `page.title()` IS A ROUND TRIP, and 

145 # that is the whole difference between a row that reports and a 

146 # row that pretends. A browser whose engine has been killed goes 

147 # on answering its url from memory, so a description built out of 

148 # urls alone reads exactly like a healthy one - measured 

149 # 2026-09-14, a status answering "page: https://..." over a 

150 # process that no longer existed. 

151 with swallow("a page that cannot say where it is answers blank"): 

152 row["url"] = page.url 

153 # A title that will not come is ordinary - a page mid-navigation 

154 # must not make the others unreadable - but a CLOSED TARGET is 

155 # the browser being gone, and it goes to the caller, which is the 

156 # only thing that can say so. 

157 with swallow("a page mid-navigation cannot say its title", 

158 unless=(TargetClosedError,)): 

159 row["title"] = await page.title() 

160 out.append(row) 

161 return out

179 #: A capture that has delivered nothing for this long is not a quiet page: 

180 #: it is a capture that has stopped. The engine delivers at WATCH_FPS 

181 #: whether or not anything on the page changed - measured 2026-09-14 on a 

182 #: page that never moves: 93 frames in 4 s, the longest gap 78 ms - so two 

183 #: seconds of silence is fifty missing frames. 

184 #: 

185 #: ⛔ A FRAME SERVED WITHOUT AN AGE IS A FROZEN PANE THAT LOOKS LIVE. The 

186 #: engine's window capture can end for good on its own: the WebRTC capturer 

187 #: it rests on reports a PERMANENT error the first time a headed window is 

188 #: minimised, the capture timer is cancelled, and nothing tells the client. 

189 #: This method then answered the last frame it held, forever, and the pane 

190 #: showed a search page while the browser was two sites further on. Now a 

191 #: frame older than this is a reason to stop and start the capture again; 

192 #: a restart that stays silent is dropped with the reason, so the pane says 

193 #: what is wrong instead of showing where the browser was.

196 async def watch_frame(self, timeout: float = 3.0) -> bytes: 

197 """The latest JPEG frame of the WINDOW the page lives in. 

198 

199 `page.screenshot()` is the content viewport and can never show the 

200 pointer, which the engine draws in the browser chrome precisely so 

201 that no page can see it. A person watching an agent work wants the 

202 pointer, the tab strip and the address bar, and that is what the 

203 engine's screencast captures: the window, through the operating 

204 system, in the parent process, with nothing injected into the page. 

205 

206 The capture is started on first use and kept running for the life of 

207 the page, so the frame answered here is at most a twenty-fifth of a 

208 second old - and if it is older than STALE_AFTER the capture is 

209 started again, because the engine does not say when one ends. 

210 """ 

211 page = self.page() 

212 key = id(page) 

213 state = self._watch.get(key) 

214 if (state is not None and state["latest"] 

215 and self._clock() - state["at"] > self.STALE_AFTER): 

216 await self._stop_watch(key) 

217 state = None 

218 if state is None: 

219 state = {"page": page, "latest": b"", "arrived": asyncio.Event(), 

220 "at": self._clock()} 

221 

222 def on_frame(frame: dict) -> None: 

223 state["latest"] = frame["data"] 

224 state["at"] = self._clock() 

225 state["arrived"].set() 

226 

227 try: 

228 await page.screencast.start(on_frame=on_frame, 

229 size=dict(self.WATCH_SIZE), 

230 fps=self.WATCH_FPS) 

231 except Exception as refused:

233 # Say which feature is missing rather than surfacing a 

234 # protocol sentence about a guid. 

235 raise RuntimeError( 

236 "the live window view needs invisible-playwright with " 

237 "page.screencast and an engine from firefox-28 on: the " 

238 "browser answered %s" % refused) from refused 

239 self._watch[key] = state 

240 if not state["latest"]: 

241 try: 

242 await asyncio.wait_for(state["arrived"].wait(), timeout) 

243 except asyncio.TimeoutError: 

244 # Dropped, not kept: a capture that delivered nothing from the 

245 # start may have died at birth - the engine ends one on a 

246 # minimised window without a word - and keeping it would make 

247 # every later look wait on a capture that cannot answer. The 

248 # next look starts its own, which succeeds the moment the window 

249 # can be captured again. 

250 await self._stop_watch(key) 

251 raise RuntimeError( 

252 "the window capture started but no frame arrived in " 

253 "%.0f s; a minimised window is captured as nothing" % timeout) 

254 return state["latest"]
