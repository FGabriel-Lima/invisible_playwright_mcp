34def json_capped(obj: Any, limit: int = DEFAULT_MAX_CHARS) -> str: 

35 """Serialize obj as JSON, capped at `limit` chars. 

36 

37 Never slices an already-serialized string, which would yield invalid JSON. 

38 When the payload is too big it returns a small, always-valid envelope. 

39 

40 For anything that is not a list of elements this is the best that can be 

41 done. `capped_elements` below is what the snapshot uses, and it exists 

42 because this envelope was throwing away the whole page. 

43 """ 

44 s = json.dumps(obj) 

45 if len(s) <= limit: 

46 return s 

47 return json.dumps({"truncated": True, "chars": len(s), "preview": s[:limit]})

50def capped_elements(head: dict, elements: list, limit: int = DEFAULT_MAX_CHARS) -> str: 

51 """As many elements as fit under `limit`, and a count of what did not. 

52 

53 The snapshot used to serialize everything and then hand back an envelope 

54 with a slice of the JSON string inside when it was too long. The slice is 

55 not parseable, so on any page above the cap the caller received zero usable 

56 elements. Not the first fifty: zero. Measured on a page with 160 elements at 

57 about 112 characters each, a 6000-character cap returned nothing at all, 

58 and a real results page passes that easily. 

59 

60 So the list is what gets shortened, in document order, and the answer says 

61 how many were left out. A partial list a model can act on beats a complete 

62 one it cannot parse. 

63 """ 

64 out = list(elements) 

65 while True: 

66 payload = dict(head) 

67 payload["interactive_elements"] = out 

68 omitted = len(elements) - len(out) 

69 if omitted: 

70 payload["omitted_elements"] = omitted 

71 payload["hint"] = "raise max_chars to see the rest" 

72 s = json.dumps(payload) 

73 if len(s) <= limit or not out: 

74 return s 

75 # Drop roughly the overflow rather than one at a time: a page with two 

76 # thousand elements would otherwise re-serialize two thousand times. 

77 overflow = len(s) - limit 

78 drop = max(1, int(len(out) * overflow / len(s)) + 1) 

79 out = out[:-drop]

98async def navigate(session, url: str, wait_until: str = "domcontentloaded") -> str: 

99 """Go to a url, and say what came back. 

100 

101 ⛔ IT USED TO ANSWER `navigated to {url}` WHATEVER HAPPENED, and both halves 

102 of that were capable of being untrue. A page that answered 404, 403 or 500 

103 got the same sentence as one that answered 200, so a model reading the reply 

104 had no way to tell a missing page from a real one and would go on to read an 

105 error document as content. And after a redirect the url in the sentence was 

106 the one ASKED FOR, not the one landed on, which is the same defect as the 

107 silent cut `read_text` used to do: a caller cannot see what it is not told. 

108 

109 Both halves are now read off the Response. The status is the server's own 

110 verdict, and the url is `response.url`, which is where the redirect chain 

111 actually ended. 

112 

113 ⛔ AND THIS IS WHY `pyproject.toml` FLOORS `invisible-playwright` AT 0.13.2, 

114 not at 0.13.0. Below that version `goto` answered `None` on every navigation 

115 ([B200] in the engine's docs), so this function would report "no HTTP 

116 response" for every page in the world - which is worse than the sentence it 

117 replaces, because it is a confident and wrong statement rather than a vague 

118 one. The floor is load-bearing, not hygiene. 

119 """ 

120 if not session.pages(): 

121 await session.new_page() 

122 page = session.page() 

123 response = await page.goto(url, wait_until=wait_until, timeout=NAVIGATION_TIMEOUT_MS) 

124 if response is None: 

125 # A same-document navigation (an anchor, or the same url again) creates 

126 # no document and so has no response. Playwright answers None here and 

127 # so do we: naming the reason keeps it from reading as a failure. 

128 return f"navigated to {page.url} (no HTTP response: same-document navigation)" 

129 return f"navigated to {response.url} (HTTP {response.status})"

260 // Across 958 elements on real pages, 88.3% carried id, name or href and 

261 // every one of those reached the right node - but only 47.6% reached it 

262 // ALONE. For the other 41% the obvious selector matches several nodes, and 

263 // Playwright acts on the first, so a model looking at the third of five 

264 // identical links clicks the first and is told it succeeded. Nothing fails 

265 // and nothing is logged: it just quietly does the wrong thing.

270 // being invisible still occupies a position in querySelectorAll.

272 function nodesFor(sel) {

275 try { n = Array.from(document.querySelectorAll(sel)); } catch (err) { n = []; }

277 } 

278 return matches.get(sel);

280 // What Playwright's CSS engine finds for `sel` searched from `host`, in the 

281 // order it finds it: the host's light descendants, then its shadow root, 

282 // then every open shadow root below either. That is `_queryCSS` with 

283 // pierceShadow, transcribed, because the index into this list is what 

284 // `>> nth=` counts against, and a different order would aim it at a 

285 // different element without failing. 

286 function piercedFrom(host, sel) { 

287 let out = []; 

288 function query(root) { 

289 out = out.concat(Array.from(root.querySelectorAll(sel))); 

290 if (root.shadowRoot) query(root.shadowRoot); 

291 for (const e of root.querySelectorAll('*')) if (e.shadowRoot) query(e.shadowRoot); 

292 } 

293 try { query(host); } catch (err) { out = []; } 

294 return out;

298 // serialized as JSON and every double quote in it would come back as two

300 function attr(s) { return String(s).replace(/\\\\/g, '\\\\\\\\').replace(/'/g, "\\\\'"); } 

301 function handle(el, href) { 

302 let base = null, fromHref = false; 

303 if (el.id) base = '#' + cssq(el.id); 

304 else if (el.name) base = el.tagName.toLowerCase() + "[name='" + attr(el.name) + "']"; 

305 else if (href) { base = "a[href='" + attr(href) + "']"; fromHref = true; } 

306 // The two below are APPENDED to the order rather than woven into it, so 

307 // no selector that already worked changes and the risk of regressing 

308 // the 88% is zero. They exist because 9.5% of elements had no handle at 

309 // all and fell back on coordinates, which go stale the moment the page 

310 // scrolls: of those, 58% carried a data-testid - an attribute whose 

311 // whole purpose is to be a stable unique handle - and a further quarter 

312 // an aria-label that was unique on the page. 

313 else { 

314 const dt = el.getAttribute('data-testid') || el.getAttribute('data-test') 

315 || el.getAttribute('data-qa') || el.getAttribute('data-cy'); 

316 if (dt) { 

317 const which = el.getAttribute('data-testid') ? 'data-testid' 

318 : el.getAttribute('data-test') ? 'data-test' 

319 : el.getAttribute('data-qa') ? 'data-qa' : 'data-cy'; 

320 base = '[' + which + "='" + attr(dt) + "']"; 

321 } else { 

322 const al = el.getAttribute('aria-label'); 

323 if (al) base = "[aria-label='" + attr(al) + "']"; 

324 } 

325 } 

326 if (!base) return null; 

327 // ⛔ INSIDE A SHADOW ROOT the document cannot see the element, so a 

328 // selector searched from the document by querySelectorAll cannot be 

329 // checked for uniqueness there, and the page's own tools would say it 

330 // matches nothing. Measured 2026-09-30 on a real form whose 

331 // every field is an <input> inside an open shadow root of a custom 

332 // element (<x-textfield id="firstName">): the snapshot listed none of 

333 // them. The handle is the HOST's handle, then `>>`, then this 

334 // element's handle searched from the host - Playwright's own chaining, 

335 // which pierces open roots - so `#firstName >> #input`. 

336 const root = el.getRootNode ? el.getRootNode() : document; 

337 if (root && root !== document && root.host) { 

338 const outer = handle(root.host, undefined); 

339 if (!outer) return null; 

340 const inner = piercedFrom(root.host, base); 

341 const k = inner.indexOf(el); 

342 if (k < 0) return null; 

343 const chained = outer.sel + ' >> ' + base; 

344 return {sel: inner.length === 1 ? chained : chained + ' >> nth=' + k, 

345 fromHref: fromHref}; 

346 } 

347 const n = nodesFor(base); 

348 if (n.length === 1) return {sel: base, fromHref: fromHref}; 

349 const i = n.indexOf(el); 

350 if (i < 0) return null; 

351 return {sel: ':nth-match(' + base + ', ' + (i + 1) + ')', fromHref: fromHref};

447async def snapshot(session, max_chars: int = 0) -> str: 

448 """Title, url, and the interactive elements that are actually visible. 

449 

450 Not the accessibility tree, and the reason is measured rather than 

451 aesthetic: on a real sign-up page a single country `<select>` contributes 

452 about two hundred `<option>` nodes, which fill the character cap before the 

453 form the caller was looking for appears at all. Filtering to elements a 

454 caller can act on - and to `offsetParent !== null`, so hidden ones do not 

455 count - keeps the answer about the page rather than about its longest 

456 dropdown. 

457 """ 

458 d = await session.page().evaluate(SNAPSHOT_JS) 

459 if not max_chars: 

460 return json.dumps(d) 

461 elements = d.pop("interactive_elements", []) 

462 return capped_elements(d, elements, limit=max_chars)

650async def _on_selector(session, selector: str, what: str, act): 

651 """Run an action aimed at a selector, and when it fails say why and what

653

658 seconds spent to learn nothing.

743def what_the_field_kept(selector: str, text: str, before: dict, after: dict) -> str: 

744 """The sentence `browser_type` answers, from what the field holds now. 

745 

746 ⛔ "TYPED INTO" IS A CLAIM ABOUT THE FIELD, NOT ABOUT THE CALL, and every 

747 other outcome is SAID rather than retried. Typing again is not idempotent: 

748 a field that turns `a@b.io,` into a chip took the text the first time, and 

749 a second pass makes a second chip. Nor is every short value a failure: a 

750 one-time code split across six boxes keeps one digit here because the page 

751 moved the focus on, which is the page working. So each shape is named for 

752 what it is, and the caller decides what follows. 

753 """ 

754 value = after.get("value") 

755 secret = bool(before.get("secret") or after.get("secret")) 

756 if value is None: 

757 return "typed into %s; it is not a field whose text can be read back" % selector 

758 if value == text: 

759 moved = "" if after.get("focused") else "; then the page moved the focus on" 

760 return "typed into %s%s" % (selector, moved) 

761 if not after.get("focused") and text.startswith(value): 

762 if not value: 

763 return ("typed toward %s, but the page moved the focus away before any " 

764 "of it stayed there; the keys went where the focus went. Read " 

765 "the page to see where." % selector) 

766 return ("typed into %s until the page moved the focus to another field after " 

767 "%d of %d characters, as a code split across boxes does: %s holds " 

768 "%s and the rest went where the focus went. Read the page to check." 

769 % (selector, len(value), len(text), selector, _shown(value, secret))) 

770 if value and text.startswith(value): 

771 return ("%s kept only the first %d of %d characters and refused the rest, " 

772 "as a maxlength does; the rest went nowhere. Shorten the text if " 

773 "the field cannot take more." % (selector, len(value), len(text))) 

774 caution = ("Read the page before typing again: a field that turns text into " 

775 "items would take it twice.") 

776 if not value: 

777 return ("%s is empty after typing: the page took the text out of the field. " 

778 "A tag or chip field does that when a comma or Enter turns it into an " 

779 "item, and a page that rewrites the field from its own state does it " 

780 "while it answers the focus. %s" % (selector, caution)) 

781 if text.endswith(value): 

782 return ("%s holds only the last %d of %d characters: the page emptied the " 

783 "field while it was being typed, as a page that rewrites a field " 

784 "from its own state does when it answers the focus or another " 

785 "field's change. %s" % (selector, len(value), len(text), caution)) 

786 return "typed into %s; the page shows it as %s" % (selector, _shown(value, secret))

927def upload_dirs(env=None) -> list[str]: 

928 """The directories named in INVISIBLE_MCP_UPLOAD_DIRS. A relative entry is 

929 refused rather than resolved against wherever the server started. 

930 

931 ⛔ AND SO IS ONE THAT RESOLVES SOMEWHERE ELSE. Resolving the entry would let 

932 whoever can replace it with a symlink choose the root: a staging directory 

933 turned into a link to its parent widens every upload to the parent's whole 

934 tree, silently. An entry whose real path is not itself (a link anywhere in 

935 it, or a directory that does not exist) turns uploads off with the reason. 

936 """

938 dirs = [] 

939 for entry in (e.strip() for e in raw.split(os.pathsep)): 

940 if not entry: 

941 continue 

942 if not os.path.isabs(entry): 

943 raise RuntimeError( 

944 f"{UPLOAD_DIRS_ENV} names {entry!r}, which is not an absolute path") 

945 named = os.path.normpath(entry) 

946 real = os.path.realpath(named)

948 # one directory on Windows, and only a link makes them two. 

949 if os.path.normcase(real) != os.path.normcase(named) or not os.path.isdir(real):

951 f"{UPLOAD_DIRS_ENV} names {entry!r}, which is not a directory at " 

952 f"that exact path (it resolves to {real!r}); uploads are off") 

953 dirs.append(real) 

954 return dirs
